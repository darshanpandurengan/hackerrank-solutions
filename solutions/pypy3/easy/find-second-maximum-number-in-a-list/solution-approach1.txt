# Technique   set-sort-indexing
# Time        O(N log N)
# Space       O(N)
# Trick       The set() constructor removes duplicates, allowing the second-to-last index [-2] of the sorted list to represent the runner-up score.
# Hint        set() is unordered; sorting is required before indexing.
# ──────────────────────────────────────────────────

if __name__ == '__main__':
    n = int(input())
    arr = map(int, input().split())
    s = list(set(arr))
    s.sort()
    print(s[-2])
